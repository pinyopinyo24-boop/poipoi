const ROOT = process.cwd();

const FORBIDDEN = [
  ".env",
  ".env.local",
  ".env.production",
  "package-lock.json",
  "pnpm-lock.yaml",
  "yarn.lock",
  ".git"
];

function safePath(filePath: string): string {
  const normalized = filePath.replace(/\\/g, "/").replace(/^\.\/+/, "");
  const absolute = path.resolve(ROOT, normalized);

  if (!absolute.startsWith(ROOT + path.sep) && absolute !== ROOT) {
    throw new Error(`プロジェクト外のファイルは禁止: ${filePath}`);
  }

  if (!fs.existsSync(absolute)) {
    throw new Error(`変更対象ファイルが存在しません: ${filePath}`);
  }

  const realRoot = fs.realpathSync(ROOT);
  const realAbsolute = fs.realpathSync(absolute);

  if (
    !realAbsolute.startsWith(realRoot + path.sep) &&
    realAbsolute !== realRoot
  ) {
    throw new Error(`実体パスがプロジェクト外です: ${filePath}`);
  }

  if (
    FORBIDDEN.some(
      x =>
        normalized === x ||
        normalized.startsWith(x + "/")
    )
  ) {
    throw new Error(`変更禁止ファイルです: ${filePath}`);
  }

  return absolute;
}

function git(args: string[]): string {
  return execFileSync("git", args, {
    cwd: ROOT,
    encoding: "utf8"
  }).trim();
}

/**
 * Phase 3: 承認済みファイルのGit安全検証
 *
 * - 承認対象は1ファイルだけ
 * - 対象外ファイルの変更をコミット対象に含めない
 * - 既存のステージ済み変更がある場合は停止
 * - Git管理外ファイルのコミットを禁止
 */
function validateApprovedGitChanges(
  approvedFiles: string[],
  targetPath: string
): void {
  const normalize = (value: string): string =>
    value.replace(/\\/g, "/");

  const files = [...new Set(approvedFiles.map(normalize))];
  const target = normalize(targetPath);

  if (files.length !== 1 || files[0] !== target) {
    throw new Error("承認対象以外のファイル変更を拒否しました");
  }

  if (
    target.startsWith("/") ||
    target.split("/").includes("..") ||
    target.includes("\0")
  ) {
    throw new Error("不正な変更対象パスです");
  }

  const staged = git(["diff", "--cached", "--name-only"]);

  if (staged.trim()) {
    throw new Error("既存のステージ済み変更があります");
  }

  const tracked = git(["ls-files", "--", target])
    .split(/\r?\n/)
    .filter(Boolean)
    .map(normalize);

  if (!tracked.includes(target)) {
    throw new Error("Git管理外のファイルは自己進化できません");
  }

  const changed = git(["status", "--porcelain", "--", target]);

  if (!changed.trim()) {
    throw new Error("承認対象ファイルに変更がありません");
  }
}
function validateSingleCommit(files: string[]): void {
  const unique = [...new Set(files.map(f => f.replace(/\\/g, "/")))];

  if (unique.length !== 1) {
    throw new Error("Gitコミット対象は1ファイルだけです");
  }

  const staged = git(["diff", "--cached", "--name-only"]).trim();

  if (staged) {
    throw new Error("既存のステージ済み変更があるため停止");
  }

  const status = git(["status", "--porcelain", "--", unique[0]]);

  if (!status.trim()) {
    throw new Error("コミット対象の差分がありません");
  }
}
function commitChanges(changedFiles: string[]): void {
  const expectedFiles = [...new Set(changedFiles)].sort();

  if (expectedFiles.length === 0) {
    throw new Error("Gitに確定する変更がありません");
  }

  const existingStaged = git(["diff", "--cached", "--name-only"])
    .split(/\r?\n/)
    .filter(Boolean);

  if (existingStaged.length > 0) {
    throw new Error(
      "既存のステージ済み変更があります。誤コミット防止のため中止します。\n" +
      existingStaged.join("\n")
    );
  }

  git(['add', ...expectedFiles]);

  const stagedFiles = git(["diff", "--cached", "--name-only"])
    .split(/\r?\n/)
    .filter(Boolean);

  const actualFiles = [...new Set(stagedFiles)].sort();

  if (JSON.stringify(actualFiles) !== JSON.stringify(expectedFiles)) {
    git(["reset", "--", ...expectedFiles]);
    throw new Error(
      "ステージ対象と変更対象が一致しません。コミットを中止しました。\n" +
      "予定: " + expectedFiles.join(", ") + "\n" +
      "実際: " + actualFiles.join(", ")
    );
  }

  try {
    git([
      "commit",
      "-m",
      "feat: apply safe poipoi self evolution",
    ]);
  } catch (commitError) {
    const commitMessage =
      commitError instanceof Error
        ? commitError.message
        : String(commitError);

    try {
      git(["reset", "--", ...expectedFiles]);
    } catch (unstageError) {
      const unstageMessage =
        unstageError instanceof Error
          ? unstageError.message
          : String(unstageError);

      throw new Error(
        "Gitコミットに失敗しました。作業中の変更は保持されています。\n" +
          "ステージ解除にも失敗しました: " + unstageMessage + "\n" +
          "コミットエラー: " + commitMessage
      );
    }

    throw new Error(
      "Gitコミットに失敗しました。作業中の変更は保持し、ステージだけ解除しました。\n" +
        commitMessage
    );
  }
}

function assertGitSafe() {
  const status = git(["status", "--porcelain"]);
  const unexpectedChanges = status
    .split(/\r?\n/)
    .filter(Boolean)
    .filter(line => line.slice(0, 2) !== "??");

  if (unexpectedChanges.length > 0) {
    throw new Error(
      "追跡対象に未コミット変更またはステージ済み変更があります。安全のため中止します。\n" +
      unexpectedChanges.join("\n")
    );
  }
}
function validateBeforeCommit(
  changedFiles: string[],
  verificationApproved = false
): void {
  if (!verificationApproved) {
    throw new Error("検証の明示的な許可がないため処理を中止します");
  }
  const result = spawnSync(
    process.platform === "win32" ? "pnpm.cmd" : "pnpm",
    ["exec", "tsc", "--noEmit", "--pretty", "false"],
    {
      cwd: process.cwd(),
      encoding: "utf8",
      shell: process.platform === "win32",
    }
  );

  if (result.status === 0) return;

  const output = [result.stdout, result.stderr, result.error?.message]
    .filter(Boolean)
    .join("\n");

  throw new Error("型チェックに失敗したためコミットを中止しました。\n" + output);
}
function extractTarget(request: string): string {
  const match = request.match(
    /((?:server|src)[\/\\][^\s"'`]+?\.(?:ts|tsx|js|jsx))/
  );

  if (!match) {
    throw new Error(
      "対象ファイルを特定できません。\n" +
      "例: server/core/SelfEvolutionWorkflow.ts を改善する"
    );
  }

  return match[1].replace(/\\/g, "/");
}

function extractJson(text: string): AIChange[] {
    const errors: string[] = [];
const trimmed = text.trim();
  const candidates: string[] = [trimmed];

  const unfenced = trimmed
    .replace(/^```(?:json)?\s*/i, "")
    .replace(/\s*```$/i, "")
    .trim();

  if (unfenced !== trimmed) candidates.push(unfenced);

  const firstArray = trimmed.indexOf("[");
  const lastArray = trimmed.lastIndexOf("]");
  if (firstArray >= 0 && lastArray > firstArray) {
    candidates.push(trimmed.slice(firstArray, lastArray + 1));
  }

  const firstObject = trimmed.indexOf("{");
  const lastObject = trimmed.lastIndexOf("}");
  if (firstObject >= 0 && lastObject > firstObject) {
    candidates.push(trimmed.slice(firstObject, lastObject + 1));
  }

  const normalize = (value: unknown, index: number): AIChange => {
    if (!value || typeof value !== "object") {
      throw new Error(`AI変更 #${index + 1} の形式が不正です`);
    }

    const item = value as Record<string, unknown>;
    if (
      typeof item.filePath !== "string" ||
      typeof item.oldCode !== "string" ||
      typeof item.newCode !== "string" ||
      typeof item.reason !== "string"
    ) {
      throw new Error(`AI変更 #${index + 1} の必須項目が不足しています`);
    }

    return {
      filePath: item.filePath,
      oldCode: item.oldCode,
      newCode: item.newCode,
      reason: item.reason,
    };
  };

  const seenCandidates = new Set<string>();
  for (let candidateIndex = 0; candidateIndex < candidates.length; candidateIndex += 1) {
    const candidate = candidates[candidateIndex];
    if (seenCandidates.has(candidate)) continue;
    seenCandidates.add(candidate);

    try {
      const parsed: unknown = JSON.parse(candidate);

      if (Array.isArray(parsed)) return parsed.map(normalize);

      if (parsed && typeof parsed === "object") {
        const value = parsed as Record<string, unknown>;
        if (Array.isArray(value.changes)) {
          return value.changes.map(normalize);
        }
        if (typeof value.response === "string") {
          const response = value.response.trim();
          candidates.push(response);

          const responseUnfenced = response
            .replace(/^```(?:json)?\s*/i, "")
            .replace(/\s*```$/i, "")
            .trim();
          if (responseUnfenced !== response) candidates.push(responseUnfenced);

          const responseFirstArray = response.indexOf("[");
          const responseLastArray = response.lastIndexOf("]");
          if (responseFirstArray >= 0 && responseLastArray > responseFirstArray) {
            candidates.push(response.slice(responseFirstArray, responseLastArray + 1));
          }

          const responseFirstObject = response.indexOf("{");
          const responseLastObject = response.lastIndexOf("}");
          if (responseFirstObject >= 0 && responseLastObject > responseFirstObject) {
            candidates.push(response.slice(responseFirstObject, responseLastObject + 1));
          }
          continue;
        }
        return [normalize(parsed, 0)];
      }
    } catch (error) {
      errors.push(error instanceof Error ? error.message : String(error));
      continue;
    }
  }

  throw new Error("AI応答に有効なJSON変更内容がありません。解析詳細: " + (errors.join(" | ") || "JSON候補を解析できません") + "。応答先頭: " + text.slice(0, 800));
}
async function invokeAI(messages: unknown[]): Promise<string> {
  const configuredUrl = process.env.LOCAL_AI_URL || "http://localhost:11434";
  const baseUrl = new URL(configuredUrl);

  if (
    baseUrl.protocol !== "http:" ||
    !["localhost", "127.0.0.1", "[::1]"].includes(baseUrl.hostname) ||
    baseUrl.username ||
    baseUrl.password ||
    baseUrl.search ||
    baseUrl.hash ||
    baseUrl.pathname !== "/"
  ) {
    throw new Error("自己進化AIはローカルOllamaのみ利用できます");
  }

  const endpoint = new URL("/v1/chat/completions", baseUrl);
  const controller = new AbortController();
  const timeout = setTimeout(() => controller.abort(), 120000);

  try {
    const response = await fetch(endpoint, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({
        model: "qwen2.5-coder:7b",
        messages,
        stream: false,
        temperature: 0.2,
        max_tokens: 2048
      }),
      signal: controller.signal
    });

    if (!response.ok) {
      throw new Error("Ollama API error: HTTP " + response.status);
    }

    const data: unknown = await response.json();

    if (
      typeof data !== "object" ||
      data === null ||
      !("choices" in data) ||
      !Array.isArray(data.choices) ||
      !data.choices[0] ||
      typeof data.choices[0].message?.content !== "string"
    ) {
      throw new Error("Ollamaから有効なAI応答を取得できませんでした");
    }

    return data.choices[0].message.content;
  } finally {
    clearTimeout(timeout);
  }
}
function findRelevantTests(targetPath: string): string[] {
  const absoluteTarget = safePath(targetPath);
  const directory = path.dirname(absoluteTarget);
  const base = path.basename(
    absoluteTarget,
    path.extname(absoluteTarget)
  );

  const candidates = [
    path.join(directory, `${base}.test.ts`),
    path.join(directory, `${base}.test.tsx`),
    path.join(directory, `${base}.spec.ts`),
    path.join(directory, `${base}.spec.tsx`)
  ];

  return candidates.filter(file => fs.existsSync(file));
}
function validateChanges(
  changes: AIChange[],
  targetPath: string
): CodeChange[] {
  const seenTargets = new Set<string>();

  // POIPOI_AI_CHANGE_GUARD: reject suspicious generated code patterns
  for (const change of changes) {
    // 不正な型は既存の map 内の検証に処理を任せる
    if (
      change === null ||
      typeof change !== "object" ||
      Array.isArray(change) ||
      typeof change.newCode !== "string"
    ) {
      continue;
    }

    const generated = change.newCode;
    if (
      /(^|\n)\s*process\.env\.[A-Z0-9_]+\s*=/.test(generated) ||
      /\beval\s*\(/.test(generated) ||
      /\bnew\s+Function\s*\(/.test(generated) ||
      /\bimport\s+.*from\s+["']child_process["']/.test(generated) ||
      /\brequire\s*\(\s*["']child_process["']\s*\)/.test(generated)
    ) {
      throw new Error(
        "AI生成コードに許可されていない危険なパターンがあります: " +
        change.filePath
      );
    }
  }
  return changes.map(item => {
    if (item === null || typeof item !== "object" || Array.isArray(item)) {
      throw new Error("変更候補がオブジェクトではありません");
    }
    if (
      typeof item.filePath !== "string" ||
      typeof item.oldCode !== "string" ||
      typeof item.newCode !== "string" ||
      !item.filePath.trim() ||
      !item.oldCode.trim()
    ) {
      throw new Error("変更候補の必須項目が欠落しているか、型が不正です");
    }

    const filePath = item.filePath.replace(/\\/g, "/");
    const absolutePath = safePath(filePath);

    if (filePath !== targetPath) {
      throw new Error(
        `変更対象外のファイルです: ${filePath}`
      );
    }

    const code = fs.readFileSync(absolutePath, "utf8");
    const sourceLineEnding = code.includes("\r\n") ? "\r\n" : "\n";
    const normalizeLineEndings = (value: string): string =>
      value.replace(/\r\n/g, "\n").replace(/\r/g, "\n");
    const normalizedOldCode = normalizeLineEndings(item.oldCode);
    const normalizedNewCode = normalizeLineEndings(item.newCode);
    const oldCode =
      sourceLineEnding === "\n"
        ? normalizedOldCode
        : normalizedOldCode.replace(/\n/g, "\r\n");
    const newCode =
      sourceLineEnding === "\n"
        ? normalizedNewCode
        : normalizedNewCode.replace(/\n/g, "\r\n");
    const targetKey = JSON.stringify([filePath, oldCode]);

    if (seenTargets.has(targetKey)) {
      throw new Error(`同じ置換対象が重複しています: ${filePath}`);
    }

    seenTargets.add(targetKey);

    if (!oldCode) {
      throw new Error(`置換対象コードが空です: ${filePath}`);
    }

    const occurrences = code.split(oldCode).length - 1;

    if (occurrences !== 1) {
      throw new Error(
        `置換対象コードを一意に特定できません: ${filePath} / ${occurrences}件`
      );
    }

    if (newCode === oldCode) {
      throw new Error(`AI生成コードに変更がありません: ${filePath}`);
    }

    const forbiddenGeneratedPatterns = [
      "export const export const",
      "export default export default",
      "import import ",
      "class class ",
      "function function ",
      "interface interface ",
      "type type ",
      "const const ",
      "let let ",
      "var var ",
    ];

    if (forbiddenGeneratedPatterns.some(pattern => newCode.includes(pattern))) {
      throw new Error(
        `AI生成コードに不正な重複構文があります: ${filePath}`
      );
    }



    if (
      newCode.includes("process.env.") ||
      newCode.includes("API_KEY") ||
      newCode.includes("SECRET")
    ) {
      throw new Error(
        `秘密情報を含む可能性がある変更を拒否しました: ${filePath}`
      );
    }

    return {
      filePath,
      oldCode,
      newCode,
      reason: item.reason
    };
  });
}

  public async runSimulation(userId: string): Promise<SimulationResult> {

/**
 * EvolutionScheduler
 *
 * SelfEvolutionWorkflow を定期的・安全に実行するスケジューラ。
 *
 * ## 既存 Heartbeat との関係
 *   - `server/_core/heartbeat.ts` の HeartbeatJob 型定義を再利用し、
 *     ジョブ記述を統一した形式で管理する。
 *   - 実際のタイマー制御はローカル `setInterval` で行う（Heartbeat スタブ API は
 *     HTTP コールバック前提のため、ローカル完結の本実装には使用しない）。
 *
 * ## 安全条件（いずれかに違反すると自動停止または execution をスキップ）
 *   1. `self_improvement.enable_auto_cycle` が true でなければ実行しない
 *   2. 1サイクルの実変更数が `maxChangesPerCycle` を超えたら execution をスキップ
 *   3. critical 候補は自動適用しない（ImprovementExecutor 側で拒否済み）
 *   4. 累積失敗率が `failureRateThreshold` を超えたら自動停止
 *   5. 連続失敗が `maxConsecutiveFailures` に達したら自動停止
 *   6. rollback_failed が 1 件でも発生したら即時停止
 *   7. 同時に複数サイクルを実行しない（`running` フラグで排他制御）
 */
export class EvolutionScheduler {
  private static instance: EvolutionScheduler;

  // ── 依存 ──
  private workflow: SelfEvolutionWorkflow;
  private applier: SafeImprovementApplier;
  private learningEngine: SelfLearningEngine;
  private memoryEngine: MemoryEngine;

  // ── 状態 ──

  private config: EvolutionSchedulerConfig;
  private runHistory: ScheduledRunResult[];
  /** setInterval の handle */
  private timer: ReturnType<typeof setInterval> | null = null;
  /** 現在サイクルが実行中かどうか（排他制御用） */
  private cycleRunning = false;
  /** 連続失敗カウンタ */
  private consecutiveFailures = 0;
  /** HeartbeatJob 形式のジョブ記述（型再利用のみ） */
  private jobDescriptor: HeartbeatJob;

  private constructor() {
    this.workflow = SelfEvolutionWorkflow.getInstance();
    this.applier = SafeImprovementApplier.getInstance();
    this.learningEngine = SelfLearningEngine.getInstance();
    this.memoryEngine = MemoryEngine.getInstance();
    this.runHistory = [];
    this.config = {
      intervalMs: DEFAULT_INTERVAL_MS,
      maxChangesPerCycle: DEFAULT_MAX_CHANGES_PER_CYCLE,
      maxConsecutiveFailures: DEFAULT_MAX_CONSECUTIVE_FAILURES,
      failureRateThreshold: 0.5,
    };
    this.jobDescriptor = {
      name: 'self-evolution',
      cron: '0 0 * * * *', // hourly (6-field: sec min hour dom mon dow)
      path: '/api/scheduled/self-evolution',
      method: 'POST',
      description: 'Periodic self-improvement cycle for PoiPoi AI',
    };
  }

  public static getInstance(): EvolutionScheduler {
    if (!EvolutionScheduler.instance) {
      EvolutionScheduler.instance = new EvolutionScheduler();

    return EvolutionScheduler.instance;

  // ── 設定 API ────────────────────────────────────────────────────────────

  /** スケジューラ設定を更新する */
  public configure(partial: Partial<EvolutionSchedulerConfig>): void {
    if (partial.intervalMs !== undefined) {
      if (!Number.isSafeInteger(partial.intervalMs) ||
          partial.intervalMs < MIN_INTERVAL_MS ||
          partial.intervalMs > 2_147_483_647) {
        throw new Error(
          `intervalMs must be between ${MIN_INTERVAL_MS}ms and 2147483647ms (got ${partial.intervalMs})`
        );
      }
    }
    if (partial.maxChangesPerCycle !== undefined &&
        (!Number.isSafeInteger(partial.maxChangesPerCycle) ||
          partial.maxChangesPerCycle < 0 ||
          partial.maxChangesPerCycle > 20)) {
      throw new Error('maxChangesPerCycle must be an integer between 0 and 20');
    }
    if (partial.maxConsecutiveFailures !== undefined &&
        (!Number.isSafeInteger(partial.maxConsecutiveFailures) ||
          partial.maxConsecutiveFailures < 1 ||
          partial.maxConsecutiveFailures > 10)) {
      throw new Error('maxConsecutiveFailures must be an integer between 1 and 10');
    }
    if (partial.failureRateThreshold !== undefined &&
        (!Number.isFinite(partial.failureRateThreshold) || partial.failureRateThreshold < 0 || partial.failureRateThreshold > 1)) {
      throw new Error('failureRateThreshold must be between 0 and 1');
    }
    this.config = { ...this.config, ...partial };
  }

  /** 現在の設定を取得する */

  /**
   * スケジューラを起動する
   * 既に running の場合は何もしない。
   */
  public start(): void {

    this.status = 'running';
    this.consecutiveFailures = 0;

    this.timer = setInterval(() => {

        // エラーは triggerCycle 内で記録済み
      });
    }, this.config.intervalMs);
  }

  /** スケジューラを手動停止する */

    this.status = 'stopped';

  /** 現在の状態を取得する */

  // ── 手動トリガー API ─────────────────────────────────────────────────────

  /**
   * サイクルを即時1回実行する（手動トリガー）
   * スケジューラが停止していても呼べる（既存の手動実行APIを壊さない）。
   */
  public async triggerOnce(userId: string): Promise<ScheduledRunResult> {
    return this.triggerCycle(userId);
  }

  // ── 内部実行ロジック ─────────────────────────────────────────────────────

  /**
   * 1サイクルの定期実行コア
   *
   * 安全条件チェック → simulation → (条件OK なら) execution → 結果保存
   */
  private async triggerCycle(userId: string): Promise<ScheduledRunResult> {
    const runId = `sched-run-${Date.now()}-${Math.random().toString(36).substr(2, 7)}`;
    const startedAt = Date.now();

    // ─ 安全条件 1: 同時実行防止 ─
    if (this.cycleRunning) {
      const skipped = this.buildResult(runId, userId, startedAt, 'skipped', 0, 0, undefined, undefined, 'Another cycle is already running');
      this.runHistory.push(skipped);
      return skipped;
    }

    // ─ 安全条件 2: auto_cycle フラグ確認 ─
    const autoEnabled = this.applier.getCurrentValue('self_improvement.enable_auto_cycle');
    if (!autoEnabled) {
      const skipped = this.buildResult(runId, userId, startedAt, 'skipped', 0, 0, undefined, undefined, 'auto_cycle is disabled');
      this.runHistory.push(skipped);
      return skipped;
    }

    this.cycleRunning = true;
    let result: ScheduledRunResult;

    try {
      result = await this.executeOneCycle(runId, userId, startedAt);
    } catch (err) {
      const error = err instanceof Error ? err.message : String(err);
      result = this.buildResult(runId, userId, startedAt, 'error', 0, 0, undefined, undefined, error);
    } finally {
      this.cycleRunning = false;
    }

    // 失敗カウンタ更新
    if (result.outcome === 'error') {
      this.consecutiveFailures++;
    } else {
      this.consecutiveFailures = 0;
    }

    // ─ 安全条件 5: 連続失敗チェック ─
    if (this.consecutiveFailures >= this.config.maxConsecutiveFailures) {
      this.autoStop('Consecutive failures exceeded threshold');
    }

    // ─ 安全条件 4: 累積失敗率チェック ─
    this.checkFailureRate(userId, result);

    // 結果を保存して履歴へ
    this.saveRunResult(userId, result);
    this.runHistory.push(result);

    return result;
  }

  /**
   * サイクルの本体処理
   */
  private async executeOneCycle(
    runId: string,
    userId: string,
    startedAt: number
  ): Promise<ScheduledRunResult> {
    // Step 1: シミュレーション
    const simResult = await this.workflow.runSimulation(userId);

    // ─ 安全条件 3: critical 候補のフィルタリング ─
    // ImprovementExecutor 側で critical+高impact は拒否済みだが、
    // スケジューラ層でも critical 候補を approved から除外する
    const safeApproved = simResult.approvedCandidates.filter(
      (c) => c.priority !== 'critical'
    );

    // ─ 安全条件 2: 実変更数上限 ─
    if (safeApproved.length > this.config.maxChangesPerCycle) {
      return this.buildResult(
        runId, userId, startedAt, 'simulation_only',
        0, 0, simResult, undefined,
        `Approved candidates (${safeApproved.length}) exceed maxChangesPerCycle (${this.config.maxChangesPerCycle})`
      );
    }

    // ─ approved が 0 件の場合はシミュレーションのみ ─
    if (safeApproved.length === 0) {
      return this.buildResult(runId, userId, startedAt, 'simulation_only', 0, 0, simResult);
    }

    // Step 2: 実行フェーズ（simulation_only に留まらない場合のみ）
    const execResult = await this.workflow.runExecution(userId, {
      ...simResult,
      approvedCandidates: safeApproved,
    });

    const changesApplied = execResult.execution.succeeded;
    const rollbacks = execResult.execution.rolledBack;

    // ─ 安全条件 6: rollback_failed で即停止 ─
    const rollbackFailed = execResult.execution.results.filter(
      (r) => r.status === 'rolled_back' && r.failureReason?.includes('rollback')
    ).length;
    if (rollbackFailed > 0) {
      this.autoStop('rollback_failed detected — immediate stop');
    }

    // Step 3: 成功した候補を SafeImprovementApplier で実際に適用
    const succeededCandidateIds = new Set(
      execResult.execution.results
        .filter((r) => r.status === 'success')
        .map((r) => r.candidateId)
    );
    const succeededCandidates = execResult.simulation.approvedCandidates.filter(
      (c) => succeededCandidateIds.has(c.id)
    );

    let applyResult: ApplyBatchSummary | undefined;
    if (succeededCandidates.length > 0) {
      const tasks = this.buildApplyTasks(succeededCandidates);
      if (tasks.length > 0) {
        applyResult = await this.applier.applyBatch(userId, tasks);

        // rollback_failed が applyBatch でも発生したら即停止
        if (applyResult.rollbackFailed > 0) {
          this.autoStop('SafeImprovementApplier rollback_failed — immediate stop');
        }
      }
    }

    return this.buildResult(
      runId, userId, startedAt, 'executed',
      changesApplied, rollbacks,
      execResult.simulation, execResult,
      undefined,
      applyResult
    );
  }

  // ── ヘルパー ─────────────────────────────────────────────────────────────

  private buildResult(
    runId: string,
    userId: string,
    startedAt: number,
    outcome: ScheduledRunResult['outcome'],
    changesApplied: number,
    rollbacks: number,
    simulationResult?: SimulationResult,
    executionResult?: ExecutionWorkflowResult,
    error?: string,
    applyResult?: ApplyBatchSummary
  ): ScheduledRunResult {
    return {
      runId,
      userId,
      startedAt,
      completedAt: Date.now(),
      outcome,
      simulationResult,
      executionResult,
      applyResult,
      error,
      changesApplied,
      rollbacks,
    };
  }

  /**
   * 成功した ImprovementCandidate を SafeImprovementApplier のタスク形式にマッピングする。
   *
   * マッピングポリシー:
   *   - 候補の description / title に許可リストのキーが含まれる場合はそのキーへ適用
   *   - 該当なしの場合は 'self_improvement.candidate_score_threshold' に
   *     estimatedImpact を 0–100 スケールでマッピング（汎用フォールバック）
   */
  private buildApplyTasks(
    candidates: ImprovementCandidate[]
  ): Array<{
    candidate: ImprovementCandidate;
    targetKey: string;
    newValue: unknown;
  }> {
    const allowedKeys = this.applier.getAllowlist().map((t) => t.key);
    const tasks: Array<{ candidate: ImprovementCandidate; targetKey: string; newValue: unknown }> = [];

    for (const candidate of candidates) {
      // critical 候補はここでも除外（二重ガード）
      if (candidate.priority === 'critical') continue;

      const text = `${candidate.title} ${candidate.description}`.toLowerCase();

      // 許可リストのキーが候補テキストに含まれる場合は直接マッピング
      const matchedKey = allowedKeys.find((k) => text.includes(k.toLowerCase()));

      if (matchedKey) {
        const target = this.applier.getAllowlist().find((t) => t.key === matchedKey)!;
        // 数値型の場合は estimatedImpact を値として使用（スケール変換）
        let newValue: unknown;
        if (target.allowedType === 'number') {
          const min = target.minValue ?? 0;
          const max = target.maxValue ?? 100;
          newValue = Math.round(min + (candidate.estimatedImpact / 100) * (max - min));
        } else if (target.allowedType === 'boolean') {
          newValue = candidate.estimatedImpact >= 50;
        } else {
          // string: そのままスキップ（自動変更しない）
          continue;
        }
        tasks.push({ candidate, targetKey: matchedKey, newValue });
      } else {
        // フォールバック: スコアしきい値を候補のスコアに基づいて調整
        const key = 'self_improvement.candidate_score_threshold';
        const currentVal = this.applier.getCurrentValue(key);
        if (typeof currentVal === 'number') {
          // 高 impact な候補が通過した場合はしきい値を緩める（より多くの候補を通す）
          const newVal = Math.max(0, Math.min(100, currentVal - Math.round(candidate.estimatedImpact * 0.1)));
          if (newVal !== currentVal) {
            tasks.push({ candidate, targetKey: key, newValue: newVal });
          }
        }
      }
    }

    // 同一キーへの重複適用を排除（最初のタスクのみ）
    const seen = new Set<string>();
    return tasks.filter((t) => {
      if (seen.has(t.targetKey)) return false;
      seen.add(t.targetKey);
      return true;
    });
  }

  /** 自動停止 */
  private autoStop(reason: string): void {
    this.clearTimer();
    this.status = 'auto_stopped';
    this.memoryEngine.setMemory(
      'system',
      `evolution_scheduler_auto_stop_${Date.now()}`,
      { reason, stoppedAt: Date.now() }
    );
  }

  private clearTimer(): void {
    if (this.timer !== null) {
      clearInterval(this.timer);
      this.timer = null;
    }
  }

  /** 累積失敗率チェック */
  private checkFailureRate(userId: string, latestResult: ScheduledRunResult): void {
    const history = this.runHistory.filter((r) => r.userId === userId || r.userId === 'system');
    history.push(latestResult);
    if (history.length < 5) return; // サンプルが少なすぎる場合はスキップ

    const recent = history.slice(-20);
    const errors = recent.filter((r) => r.outcome === 'error').length;
    const rate = errors / recent.length;

    if (rate >= this.config.failureRateThreshold) {
      this.autoStop(`Failure rate ${(rate * 100).toFixed(1)}% exceeds threshold`);
    }
  }

  /** 実行結果を MemoryEngine と SelfLearningEngine に保存 */
  private saveRunResult(userId: string, result: ScheduledRunResult): void {
    this.memoryEngine.setMemory(
      userId,
      `scheduled_run_${result.runId}`,
      result
    );
  }

  // ── 照会 API ─────────────────────────────────────────────────────────────

  /** 実行履歴を取得する */
  public getRunHistory(limit = 20): ScheduledRunResult[] {
    return this.runHistory.slice(-limit);
  }

  /** 統計情報 */
  public getStats(): {
    totalRuns: number;
    executedRuns: number;
    simulationOnlyRuns: number;
    skippedRuns: number;
    errorRuns: number;
    totalChangesApplied: number;
    totalRollbacks: number;
    consecutiveFailures: number;
    status: SchedulerStatus;
  } {
    const h = this.runHistory;
    return {
      totalRuns: h.length,
      executedRuns: h.filter((r) => r.outcome === 'executed').length,
      simulationOnlyRuns: h.filter((r) => r.outcome === 'simulation_only').length,
      skippedRuns: h.filter((r) => r.outcome === 'skipped').length,
      errorRuns: h.filter((r) => r.outcome === 'error').length,
      totalChangesApplied: h.reduce((s, r) => s + r.changesApplied, 0),
      totalRollbacks: h.reduce((s, r) => s + r.rollbacks, 0),
      consecutiveFailures: this.consecutiveFailures,
      status: this.status,
    };
  }

  /** テスト用: 全データクリア・状態リセット */
  public resetForTest(): void {
    this.clearTimer();
    this.status = 'idle';
    this.cycleRunning = false;
    this.consecutiveFailures = 0;
    this.runHistory = [];
    this.config = {
      intervalMs: DEFAULT_INTERVAL_MS,
      maxChangesPerCycle: DEFAULT_MAX_CHANGES_PER_CYCLE,
      maxConsecutiveFailures: DEFAULT_MAX_CONSECUTIVE_FAILURES,
      failureRateThreshold: 0.5,
    };
  }
}

/**

 * 処理フロー（1候補あたり）:
 *   1. 実行前の状態を MemoryEngine にスナップショット保存
 *   2. 改善処理をシミュレーション実行（設定パラメータ調整相当）
 *   3. 成功 → 結果を MemoryEngine / SelfLearningEngine に保存
 *   4. 失敗 → スナップショットからロールバック
 *
 * 制約:
 *   - 完全ローカル動作（外部API不使用）
 *   - 危険な自動コード変更は行わない
 *   - 新規依存パッケージなし
 */
export class ImprovementExecutor {
  private static instance: ImprovementExecutor;
  private learningEngine: SelfLearningEngine;
  private memoryEngine: MemoryEngine;
  /** 実行履歴 */
  private executionHistory: ExecutionResult[];
  /** スナップショットストア */
  private snapshots: Map<string, ExecutionSnapshot>;

  private constructor() {
    this.learningEngine = SelfLearningEngine.getInstance();
    this.memoryEngine = MemoryEngine.getInstance();
    this.executionHistory = [];
    this.snapshots = new Map();
  }

  public static getInstance(): ImprovementExecutor {
    if (!ImprovementExecutor.instance) {
      ImprovementExecutor.instance = new ImprovementExecutor();
    }
    return ImprovementExecutor.instance;
  }

  // ── 公開API ─────────────────────────────────────────────────────────────

  /**
   * 単一候補を実行する
   */
  public async execute(
    userId: string,
    candidate: ImprovementCandidate
  ): Promise<ExecutionResult> {
    const startedAt = Date.now();

    // Step 1: 実行前スナップショット保存
    const snapshot = await this.takeSnapshot(userId, candidate);

    let status: ExecutionStatus = 'failure';
    let failureReason: string | undefined;
    let rolledBack = false;

    try {
      // Step 2: 改善処理のシミュレーション実行
      await this.simulateImprovement(userId, candidate);
      status = 'success';
    } catch (err) {
      // Step 3: 失敗時はロールバック
      failureReason = err instanceof Error ? err.message : String(err);
      await this.rollback(userId, candidate, snapshot);
      rolledBack = true;
      status = 'rolled_back';
    }

    const result: ExecutionResult = {
      candidateId: candidate.id,
      userId,
      status,
      durationMs: Date.now() - startedAt,
      failureReason,
      rolledBack,
      executedAt: startedAt,
    };

    // Step 4: 結果を保存
    await this.saveResult(userId, candidate, result);

    this.executionHistory.push(result);
    return result;
  }

  /**
   * 複数候補をまとめて実行する（バッチ）
   */
  public async executeBatch(
    userId: string,
    candidates: ImprovementCandidate[]
  ): Promise<ExecutionBatchSummary> {
    const results: ExecutionResult[] = [];

    for (const candidate of candidates) {
      const result = await this.execute(userId, candidate);
      results.push(result);
    }

    const summary: ExecutionBatchSummary = {
      userId,
      total: results.length,
      succeeded: results.filter((r) => r.status === 'success').length,
      failed: results.filter((r) => r.status === 'failure').length,
      rolledBack: results.filter((r) => r.rolledBack).length,
      skipped: results.filter((r) => r.status === 'skipped').length,
      results,
      executedAt: Date.now(),
    };

    // バッチサマリーをメモリに保存
    const batchKey = `execution_batch_${userId}_${summary.executedAt}`;
    this.memoryEngine.setMemory(userId, batchKey, summary);

    return summary;
  }

  /**
   * 実行履歴を取得
   */
  public getExecutionHistory(userId: string, limit = 50): ExecutionResult[] {
    return this.executionHistory
      .filter((r) => r.userId === userId)
      .slice(-limit);
  }

  /**
   * スナップショットを取得
   */
  public getSnapshot(candidateId: string): ExecutionSnapshot | undefined {
    return this.snapshots.get(candidateId);
  }

  /**
   * 統計情報
   */
  public getStats(userId: string): {
    totalExecuted: number;
    successCount: number;
    failureCount: number;
    rollbackCount: number;
    successRate: number;
  } {
    const history = this.executionHistory.filter((r) => r.userId === userId);
    const successCount = history.filter((r) => r.status === 'success').length;
    const failureCount = history.filter(
      (r) => r.status === 'failure' || r.status === 'rolled_back'
    ).length;
    const rollbackCount = history.filter((r) => r.rolledBack).length;
    const total = history.length;

    return {
      totalExecuted: total,
      successCount,
      failureCount,
      rollbackCount,
      successRate: total === 0 ? 0 : Math.round((successCount / total) * 100),
    };
  }

  /** テスト用: 全データクリア */
  public clearAllData(): void {
    this.executionHistory = [];
    this.snapshots.clear();
  }

  // ── プライベート実装 ──────────────────────────────────────────────────────

  /**
   * 実行前の状態スナップショットを作成して保存
   */
  private async takeSnapshot(
    userId: string,
    candidate: ImprovementCandidate
  ): Promise<ExecutionSnapshot> {
    const snapshot: ExecutionSnapshot = {
      candidateId: candidate.id,
      userId,
      takenAt: Date.now(),
      state: {
        candidateTitle: candidate.title,
        candidatePriority: candidate.priority,
        candidateSource: candidate.source,
        estimatedImpact: candidate.estimatedImpact,
      },
    };

    this.snapshots.set(candidate.id, snapshot);
    this.memoryEngine.setMemory(userId, `snapshot_${candidate.id}`, snapshot);

    return snapshot;
  }

  /**
   * 改善処理のシミュレーション
   *
   * ポリシー:
   *   - 実際のコード書き換えは行わない（安全保証）
   *   - 優先度・ソースに応じた改善効果をパラメータとして記録する
   *   - 「critical」かつ impact>70 の組み合わせは過剰リスクとして例外を投げる
   *     （ロールバック経路のテスト用ルールとしても機能する）
   */
  private async simulateImprovement(
    userId: string,
    candidate: ImprovementCandidate
  ): Promise<void> {
    // 過剰リスク候補は実行を拒否してロールバックをトリガー
    if (candidate.priority === 'critical' && candidate.estimatedImpact > 70) {
      throw new Error(
        `Execution rejected: candidate "${candidate.id}" has critical priority ` +
          `with high impact (${candidate.estimatedImpact}). Manual review required.`
      );
    }

    // 改善パラメータを計算（設定最適化のシミュレーション）
    const improvementParams: Record<string, unknown> = {
      candidateId: candidate.id,
      source: candidate.source,
      priority: candidate.priority,
      estimatedImpact: candidate.estimatedImpact,
      appliedAt: Date.now(),
      // 優先度別の調整値（実際のコード変更なし）
      adjustmentFactor:
        candidate.priority === 'high'
          ? 1.2
          : candidate.priority === 'medium'
            ? 1.1
            : 1.0,
    };

    // 改善パラメータをメモリに記録
    this.memoryEngine.setMemory(
      userId,
      `improvement_params_${candidate.id}`,
      improvementParams
    );
  }

  /**
   * スナップショットを使ってロールバック
   */
  private async rollback(
    userId: string,
    candidate: ImprovementCandidate,
    snapshot: ExecutionSnapshot
  ): Promise<void> {
    // 適用されたパラメータを削除してスナップショット状態を復元
    this.memoryEngine.deleteMemory(userId, `improvement_params_${candidate.id}`);

    // ロールバック記録をメモリに保存
    this.memoryEngine.setMemory(userId, `rollback_${candidate.id}`, {
      candidateId: candidate.id,
      rolledBackAt: Date.now(),
      restoredFrom: snapshot.takenAt,
    });
  }

  /**
   * 実行結果を MemoryEngine と SelfLearningEngine に保存
   */
  private async saveResult(
    userId: string,
    candidate: ImprovementCandidate,
    result: ExecutionResult
  ): Promise<void> {
    // MemoryEngine に保存
    this.memoryEngine.setMemory(
      userId,
      `execution_result_${candidate.id}`,
      result
    );

    // SelfLearningEngine に学習データとして記録
    // （学習サイクルが起動済みかどうかに依存しないよう try/catch で保護）
    try {
      const cycles = await this.learningEngine.getAllLearningCycles(userId);
      const activeCycle = cycles.find((c) => c.status === 'running' as any);

      if (activeCycle) {
        await this.learningEngine.recordDataPoint(
          userId,
          activeCycle.id,
          { candidateId: candidate.id, title: candidate.title, source: candidate.source },
          { status: result.status, durationMs: result.durationMs, rolledBack: result.rolledBack },
          result.status === 'success' ? 'success' : 'failure',
          result.durationMs,
          { priority: candidate.priority, estimatedImpact: candidate.estimatedImpact }
        );
      }
    } catch {
      // 学習サイクルが存在しない場合はスキップ（実行自体は成功）
    }
  }
}

===== server/core/SafeImprovementApplier.ts =====
import { ImprovementCandidate } from './SelfImprovementManager';
import { SelfLearningEngine } from './learning/SelfLearningEngine';
import { MemoryEngine } from './memory/MemoryEngine';

// ── 許可リスト型 ──────────────────────────────────────────────────────────

/**
 * 安全な変更対象のカテゴリ
 *
 * - 'config'   : 自己改善用の設定値（数値・文字列パラメータ）
 * - 'rule'     : 自己改善ルール（フィルタ条件・しきい値）
 * - 'prompt'   : プロンプト設定（テキスト定義）
 * - 'param'    : その他、明示的に許可登録されたパラメータ
 */
export type TargetCategory = 'config' | 'rule' | 'prompt' | 'param';

/**
 * SafeImprovementTarget — 許可された変更対象の定義
 */
export interface SafeImprovementTarget {
  /** 一意キー（例: "self_improvement.threshold" ） */
  key: string;
  category: TargetCategory;
  description: string;
  /** 現在の値 */
  currentValue: unknown;
  /** 許可される値の型 */
  allowedType: 'number' | 'string' | 'boolean';
  /** 数値の場合の最小値 */
  minValue?: number;
  /** 数値の場合の最大値 */
  maxValue?: number;
  /** 文字列の場合の最大長 */
  maxLength?: number;
}

// ── 適用結果型 ────────────────────────────────────────────────────────────

/**
 * 実際の変更適用ステータス
 *
 * - 'applied'      : 変更が確定された
 * - 'rolled_back'  : テスト失敗により元の値に戻した
 * - 'rejected'     : 許可リスト未登録 or バリデーション失敗で実行しなかった
 * - 'rollback_failed' : ロールバック自体が失敗した（安全記録のみ行う）
 */
export type ApplyStatus = 'applied' | 'rolled_back' | 'rejected' | 'rollback_failed';

/**
 * 一変更の適用結果
 */
export interface ApplyResult {
  targetKey: string;
  candidateId: string;
  userId: string;
  status: ApplyStatus;
  previousValue: unknown;
  newValue: unknown;
  /** 失敗・拒否理由 */
  reason?: string;
  durationMs: number;
  appliedAt: number;
}

/**
 * バッチ適用サマリー
 */
export interface ApplyBatchSummary {
  userId: string;
  total: number;
  applied: number;
  rolledBack: number;
  rejected: number;
  rollbackFailed: number;
  results: ApplyResult[];
  executedAt: number;
}

// ── SafeImprovementApplier ────────────────────────────────────────────────

/**
 * SafeImprovementApplier
 *
 * 許可リストに登録されたパラメータのみを実際に変更する。
 * ImprovementExecutor のシミュレーション機能とは独立した「実変更」レイヤー。
 *
 * ## 分離の原則
 * | クラス                  | 役割                                      |
 * |-------------------------|-------------------------------------------|
 * | ImprovementExecutor     | シミュレーション（パラメータ評価のみ）    |
 * | SafeImprovementApplier  | 許可リスト内パラメータの実変更 + rollback |
 *
 * ## 禁止事項（コード上で保証）
 * - 任意コードの実行
 * - ファイルへの書き込み
 * - シェルコマンド実行
 * - 外部API 呼び出し
 * - 許可リスト外のキー変更
 */
export class SafeImprovementApplier {
  private static instance: SafeImprovementApplier;

  /** 許可リスト: key → SafeImprovementTarget */
  private allowlist: Map<string, SafeImprovementTarget>;
  /** 適用履歴 */
  private applyHistory: ApplyResult[];
  /** 依存エンジン */
  private memoryEngine: MemoryEngine;
  private learningEngine: SelfLearningEngine;

  private constructor() {
    this.allowlist = new Map();
    this.applyHistory = [];
    this.memoryEngine = MemoryEngine.getInstance();
    this.learningEngine = SelfLearningEngine.getInstance();
    this.registerDefaults();
  }

  public static getInstance(): SafeImprovementApplier {
    if (!SafeImprovementApplier.instance) {
      SafeImprovementApplier.instance = new SafeImprovementApplier();
    }
    return SafeImprovementApplier.instance;
  }

  // ── 許可リスト管理 ───────────────────────────────────────────────────────

  /**
   * 対象を許可リストに登録する
   */
  public registerTarget(target: SafeImprovementTarget): void {
    this.allowlist.set(target.key, { ...target });
  }

  /**
   * 許可リストから対象を削除する
   */
  public unregisterTarget(key: string): void {
    this.allowlist.delete(key);
  }

  /**
   * 許可リストを取得する（読み取り専用コピー）
   */
  public getAllowlist(): SafeImprovementTarget[] {
    return Array.from(this.allowlist.values()).map((t) => ({ ...t }));
  }

  /**
   * 対象が許可リストに存在するか確認する
   */
  public isAllowed(key: string): boolean {
    return this.allowlist.has(key);
  }

  /**
   * 対象の現在値を取得する
   */
  public getCurrentValue(key: string): unknown {
    return this.allowlist.get(key)?.currentValue;
  }

  // ── 実変更 API ───────────────────────────────────────────────────────────

  /**
   * 改善候補から対象キーを特定し、値を実際に変更する
   *
   * @param userId     操作ユーザー
   * @param candidate  SelfImprovementManager が生成した改善候補
   * @param targetKey  変更対象のキー（許可リストに登録済みであること）
   * @param newValue   適用する新しい値
   * @param validator  オプション: 変更後に呼ばれる検証関数（テスト代わり）
   */
  public async apply(
    userId: string,
    candidate: ImprovementCandidate,
    targetKey: string,
    newValue: unknown,
    validator?: () => boolean | Promise<boolean>
  ): Promise<ApplyResult> {
    const startedAt = Date.now();

    // Step 1: 許可リスト確認
    const target = this.allowlist.get(targetKey);
    if (!target) {
      const result = this.buildResult(
        targetKey,
        candidate.id,
        userId,
        'rejected',
        undefined,
        newValue,
        `Target key "${targetKey}" is not in the allowlist`,
        Date.now() - startedAt,
        startedAt
      );
      this.applyHistory.push(result);
      return result;
    }

    // Step 2: バリデーション
    const validationError = this.validate(target, newValue);
    if (validationError) {
      const result = this.buildResult(
        targetKey,
        candidate.id,
        userId,
        'rejected',
        target.currentValue,
        newValue,
        validationError,
        Date.now() - startedAt,
        startedAt
      );
      this.applyHistory.push(result);
      return result;
    }

    // Step 3: 変更前の値を保存
    const previousValue = target.currentValue;
    this.saveBackup(userId, targetKey, previousValue);

    // Step 4: 実際に値を変更
    target.currentValue = newValue;

    // Step 5: バリデーター（テスト）を実行
    let testPassed = true;
    try {
      if (validator) {
        testPassed = await validator();
      }
    } catch {
      testPassed = false;
    }

    if (testPassed) {
      // Step 6: 成功 → 変更を確定
      const result = this.buildResult(
        targetKey,
        candidate.id,
        userId,
        'applied',
        previousValue,
        newValue,
        undefined,
        Date.now() - startedAt,
        startedAt
      );
      await this.saveResult(userId, candidate, result);
      this.applyHistory.push(result);
      return result;
    } else {
      // Step 7: 失敗 → ロールバック
      let status: ApplyStatus = 'rolled_back';
      let reason = 'Validator returned false after applying change';

      try {
        target.currentValue = previousValue;
        this.restoreBackup(userId, targetKey);
      } catch (rbErr) {
        // ロールバック自体が失敗した場合: 安全記録のみ
        status = 'rollback_failed';
        reason =
          `Validator failed AND rollback failed: ` +
          (rbErr instanceof Error ? rbErr.message : String(rbErr));
        // ロールバック失敗を MemoryEngine に記録（安全処理）
        this.memoryEngine.setMemory(userId, `rollback_failure_${targetKey}_${startedAt}`, {
          targetKey,
          candidateId: candidate.id,
          failedAt: Date.now(),
          reason,
        });
      }

      const result = this.buildResult(
        targetKey,
        candidate.id,
        userId,
        status,
        previousValue,
        newValue,
        reason,
        Date.now() - startedAt,
        startedAt
      );
      await this.saveResult(userId, candidate, result);
      this.applyHistory.push(result);
      return result;
    }
  }

  /**
   * 複数候補に対して一括適用する
   *
   * @param tasks  各タスク: { candidate, targetKey, newValue, validator }
   */
  public async applyBatch(
    userId: string,
    tasks: Array<{
      candidate: ImprovementCandidate;
      targetKey: string;
      newValue: unknown;
      validator?: () => boolean | Promise<boolean>;
    }>
  ): Promise<ApplyBatchSummary> {
    const results: ApplyResult[] = [];

    for (const task of tasks) {
      const result = await this.apply(
        userId,
        task.candidate,
        task.targetKey,
        task.newValue,
        task.validator
      );
      results.push(result);
    }

    const summary: ApplyBatchSummary = {
      userId,
      total: results.length,
      applied: results.filter((r) => r.status === 'applied').length,
      rolledBack: results.filter((r) => r.status === 'rolled_back').length,
      rejected: results.filter((r) => r.status === 'rejected').length,
      rollbackFailed: results.filter((r) => r.status === 'rollback_failed').length,
      results,
      executedAt: Date.now(),
    };

    this.memoryEngine.setMemory(userId, `apply_batch_${userId}_${summary.executedAt}`, summary);
    return summary;
  }

  // ── 照会 API ─────────────────────────────────────────────────────────────

  /**
   * 適用履歴を取得する
   */
  public getApplyHistory(userId: string, limit = 50): ApplyResult[] {
    return this.applyHistory.filter((r) => r.userId === userId).slice(-limit);
  }

  /**
   * 統計情報
   */
  public getStats(userId: string): {
    totalApplied: number;
    totalRolledBack: number;
    totalRejected: number;
    totalRollbackFailed: number;
    successRate: number;
  } {
    const history = this.applyHistory.filter((r) => r.userId === userId);
    const applied = history.filter((r) => r.status === 'applied').length;
    const rolledBack = history.filter((r) => r.status === 'rolled_back').length;
    const rejected = history.filter((r) => r.status === 'rejected').length;
    const rollbackFailed = history.filter((r) => r.status === 'rollback_failed').length;
    const total = history.length;

    return {
      totalApplied: applied,
      totalRolledBack: rolledBack,
      totalRejected: rejected,
      totalRollbackFailed: rollbackFailed,
      successRate: total === 0 ? 0 : Math.round((applied / total) * 100),
    };
  }

  /** テスト用: 全データクリア（履歴・許可リストはデフォルトを除いて残す） */
  public clearHistory(): void {
    this.applyHistory = [];
  }

  /** テスト用: 許可リストごと完全クリアして再登録 */
  public resetToDefaults(): void {
    this.allowlist.clear();
    this.applyHistory = [];
    this.registerDefaults();
  }

  // ── プライベート ──────────────────────────────────────────────────────────

  /**
   * デフォルト許可リストを登録する
   * （自己改善システム自身の設定パラメータのみ）
   */
  private registerDefaults(): void {
    const defaults: SafeImprovementTarget[] = [
      {
        key: 'self_improvement.candidate_score_threshold',
        category: 'config',
        description: '改善候補のテスト通過最低スコア（0–100）',
        currentValue: 50,
        allowedType: 'number',
        minValue: 0,
        maxValue: 100,
      },
      {
        key: 'self_improvement.max_candidates_per_cycle',
        category: 'config',
        description: '1サイクルあたりの最大候補数',
        currentValue: 20,
        allowedType: 'number',
        minValue: 1,
        maxValue: 100,
      },
      {
        key: 'self_improvement.cycle_history_limit',
        category: 'config',
        description: '保持するサイクル履歴の上限',
        currentValue: 10,
        allowedType: 'number',
        minValue: 1,
        maxValue: 200,
      },
      {
        key: 'self_improvement.failure_rate_alert_threshold',
        category: 'rule',
        description: '失敗率アラートのしきい値（0.0–1.0）',
        currentValue: 0.3,
        allowedType: 'number',
        minValue: 0,
        maxValue: 1,
      },
      {
        key: 'self_improvement.system_prompt_suffix',
        category: 'prompt',
        description: 'システムプロンプトへの付加テキスト',
        currentValue: '',
        allowedType: 'string',
        maxLength: 500,
      },
      {
        key: 'self_improvement.enable_auto_cycle',
        category: 'param',
        description: '自動改善サイクルの有効/無効',
        currentValue: false,
        allowedType: 'boolean',
      },
    ];

    for (const t of defaults) {
      this.allowlist.set(t.key, t);
    }
  }

  /**
   * 新しい値がターゲット定義に適合するかバリデーション
   * @returns エラーメッセージ（問題なければ null）
   */
  private validate(target: SafeImprovementTarget, newValue: unknown): string | null {
    // 型チェック
    if (typeof newValue !== target.allowedType) {
      return (
        `Type mismatch for "${target.key}": ` +
        `expected ${target.allowedType}, got ${typeof newValue}`
      );
    }

    // 数値範囲チェック
    if (target.allowedType === 'number') {
      const n = newValue as number;
      if (!isFinite(n)) return `Value for "${target.key}" must be a finite number`;
      if (target.minValue !== undefined && n < target.minValue) {
        return `Value ${n} is below minimum ${target.minValue} for "${target.key}"`;
      }
      if (target.maxValue !== undefined && n > target.maxValue) {
        return `Value ${n} exceeds maximum ${target.maxValue} for "${target.key}"`;
      }
    }

    // 文字列長チェック
    if (target.allowedType === 'string' && target.maxLength !== undefined) {
      const s = newValue as string;
      if (s.length > target.maxLength) {
        return (
          `String length ${s.length} exceeds maxLength ${target.maxLength} for "${target.key}"`
        );
      }
    }

    return null;
  }

  /** 変更前の値をバックアップ保存 */
  private saveBackup(userId: string, key: string, value: unknown): void {
    this.memoryEngine.setMemory(userId, `backup_${key}`, { key, value, savedAt: Date.now() });
  }

  /** バックアップから値を復元（MemoryEngine 側のクリーンアップ） */
  private restoreBackup(userId: string, key: string): void {
    this.memoryEngine.deleteMemory(userId, `backup_${key}`);
  }

  /** ApplyResult を組み立てる */
  private buildResult(
    targetKey: string,
    candidateId: string,
    userId: string,
    status: ApplyStatus,
    previousValue: unknown,
    newValue: unknown,
    reason: string | undefined,
    durationMs: number,
    appliedAt: number
  ): ApplyResult {
    return {
      targetKey,
      candidateId,
      userId,
      status,
      previousValue,
      newValue,
      reason,
      durationMs,
      appliedAt,
    };
  }

  /** 結果を MemoryEngine と SelfLearningEngine に保存 */
  private async saveResult(
    userId: string,
    candidate: ImprovementCandidate,
    result: ApplyResult
  ): Promise<void> {
    // MemoryEngine に保存
    this.memoryEngine.setMemory(
      userId,
      `apply_result_${candidate.id}_${result.appliedAt}`,
      result
    );

    // SelfLearningEngine に記録（アクティブなサイクルがあれば）
    try {
      const cycles = await this.learningEngine.getAllLearningCycles(userId);
      const activeCycle = cycles.find((c) => (c.status as string) === 'running');
      if (activeCycle) {
        await this.learningEngine.recordDataPoint(
          userId,
          activeCycle.id,
          {
            type: 'safe_apply',
            targetKey: result.targetKey,
            candidateId: candidate.id,
          },
          {
            status: result.status,
            previousValue: result.previousValue,
            newValue: result.newValue,
          },
          result.status === 'applied' ? 'success' : 'failure',
          result.durationMs,
          { category: this.allowlist.get(result.targetKey)?.category }
        );
      }
    } catch {
      // 学習サイクルなし → スキップ
    }
  }
}

export class SelfLearningEngine {

  private constructor() {

  public static getInstance(): SelfLearningEngine {

  }

  /**
   * Start learning cycle
   */
  public async startLearningCycle(userId: string): Promise<LearningCycle> {
    // Security check
    const hasPermission = await this.securityEngine.checkAuthorization(userId, 'learning:write');
    if (!hasPermission) {
      throw new Error('User does not have permission to start learning cycle');
    }

    const cycle: LearningCycle = {
      id: `cycle-${Date.now()}-${Math.random().toString(36).substr(2, 9)}`,
      userId,
      startTime: Date.now(),
      status: LearningCycleStatus.RUNNING,
      dataPoints: [],
      patterns: [],
      improvements: [],
    };

    this.learningCycles.set(cycle.id, cycle);

    // Log to security
    await this.securityEngine.logSecurityEvent('LEARNING_CYCLE_START', userId, {
      cycleId: cycle.id,
    });

    return cycle;
  }

  /**
   * Record learning data point
   */
  public async recordDataPoint(
    userId: string,
    cycleId: string,
    input: any,
    output: any,
    result: 'success' | 'failure',
    duration: number,
    metadata?: Record<string, any>
  ): Promise<LearningData> {
    // Security check
    const hasPermission = await this.securityEngine.checkAuthorization(userId, 'learning:write');
    if (!hasPermission) {
      throw new Error('User does not have permission to record learning data');
    }

    const cycle = this.learningCycles.get(cycleId);
    if (!cycle) {
      throw new Error('Learning cycle not found');
    }

    const dataPoint: LearningData = {
      id: `data-${Date.now()}-${Math.random().toString(36).substr(2, 9)}`,
      userId,
      cycleId,
      timestamp: Date.now(),
      input,
      output,
      result,
      duration,
      metadata: metadata || {},
    };

    this.learningData.push(dataPoint);
    cycle.dataPoints.push(dataPoint);

    return dataPoint;
  }

  /**
   * Complete learning cycle
   */
  public async completeLearningCycle(userId: string, cycleId: string): Promise<LearningResult> {
    // Security check
    const hasPermission = await this.securityEngine.checkAuthorization(userId, 'learning:write');
    if (!hasPermission) {
      throw new Error('User does not have permission to complete learning cycle');
    }

    const cycle = this.learningCycles.get(cycleId);
    if (!cycle) {
      throw new Error('Learning cycle not found');
    }

    cycle.endTime = Date.now();
    cycle.status = LearningCycleStatus.COMPLETED;

    // Extract patterns
    const patterns = this.extractPatterns(cycle);
    cycle.patterns = patterns;

    // Generate result
    const result: LearningResult = {
      cycleId,
      totalDataPoints: cycle.dataPoints.length,
      successRate: this.calculateSuccessRate(cycle),
      failureRate: this.calculateFailureRate(cycle),
      patterns,
      recommendations: this.generateRecommendations(cycle),
      generatedAt: Date.now(),
    };

    // Save to memory
    await this.memoryEngine.setMemory(userId, `learning_result_${cycleId}`, result);

    // Save patterns as knowledge (deferred for now)
    // Will be implemented in Knowledge integration phase

    // Log to security
    await this.securityEngine.logSecurityEvent('LEARNING_CYCLE_COMPLETE', userId, {
      cycleId,
      successRate: result.successRate,
      patternCount: patterns.length,
    });

    return result;
  }

  /**
   * Extract patterns from learning cycle
   */
  private extractPatterns(cycle: LearningCycle): Pattern[] {
    const patterns: Pattern[] = [];

    // Count successes and failures
    const successCount = cycle.dataPoints.filter((d) => d.result === 'success').length;
    const failureCount = cycle.dataPoints.filter((d) => d.result === 'failure').length;

    // Success pattern
    if (successCount > 0) {
      patterns.push({
        id: `pattern-success-${Date.now()}`,
        type: 'success',
        frequency: successCount,
        confidence: successCount / cycle.dataPoints.length,
        description: `Successfully completed ${successCount} operations`,
        examples: cycle.dataPoints
          .filter((d) => d.result === 'success')
          .slice(0, 3)
          .map((d) => d.input),
      });
    }

    // Failure pattern
    if (failureCount > 0) {
      patterns.push({
        id: `pattern-failure-${Date.now()}`,
        type: 'failure',
        frequency: failureCount,
        confidence: failureCount / cycle.dataPoints.length,
        description: `Failed ${failureCount} operations`,
        examples: cycle.dataPoints
          .filter((d) => d.result === 'failure')
          .slice(0, 3)
          .map((d) => d.input),
      });
    }

    // Performance improvement pattern
    if (cycle.dataPoints.length > 1) {
      const durations = cycle.dataPoints.map((d) => d.duration);
      const avgDuration = durations.reduce((a, b) => a + b, 0) / durations.length;
      const lastDuration = durations[durations.length - 1];

      if (lastDuration < avgDuration * 0.8) {
        patterns.push({
          id: `pattern-improvement-${Date.now()}`,
          type: 'improvement',
          frequency: 1,
          confidence: 0.8,
          description: `Performance improved by ${Math.round((1 - lastDuration / avgDuration) * 100)}%`,
          examples: [{ avgDuration, lastDuration }],
        });
      }
    }

    return patterns;
  }

  /**
   * Calculate success rate
   */
  private calculateSuccessRate(cycle: LearningCycle): number {
    if (cycle.dataPoints.length === 0) return 0;
    const successCount = cycle.dataPoints.filter((d) => d.result === 'success').length;
    return successCount / cycle.dataPoints.length;
  }

  /**
   * Calculate failure rate
   */
  private calculateFailureRate(cycle: LearningCycle): number {
    if (cycle.dataPoints.length === 0) return 0;
    const failureCount = cycle.dataPoints.filter((d) => d.result === 'failure').length;
    return failureCount / cycle.dataPoints.length;
  }

  /**
   * Generate recommendations
   */
  private generateRecommendations(cycle: LearningCycle): string[] {
    const recommendations: string[] = [];

    const successRate = this.calculateSuccessRate(cycle);
    const failureRate = this.calculateFailureRate(cycle);

    if (failureRate > 0.3) {
      recommendations.push('High failure rate detected - review error patterns');
    }

    if (successRate > 0.8) {
      recommendations.push('Strong success pattern - continue current approach');
    }

    if (cycle.dataPoints.length > 0) {
      const avgDuration =
        cycle.dataPoints.reduce((sum, d) => sum + d.duration, 0) / cycle.dataPoints.length;
      if (avgDuration > 1000) {
        recommendations.push('Performance optimization recommended');
      }
    }

    if (recommendations.length === 0) {
      recommendations.push('Continue monitoring and learning');
    }

    return recommendations;
  }

  /**
   * Get learning cycle
   */
  public async getLearningCycle(userId: string, cycleId: string): Promise<LearningCycle | null> {
    // Security check
    const hasPermission = await this.securityEngine.checkAuthorization(userId, 'learning:read');
    if (!hasPermission) {
      throw new Error('User does not have permission to read learning data');
    }

    return this.learningCycles.get(cycleId) || null;
  }

  /**
   * Get all learning cycles
   */
  public async getAllLearningCycles(userId: string): Promise<LearningCycle[]> {
    // Security check
    const hasPermission = await this.securityEngine.checkAuthorization(userId, 'learning:read');
    if (!hasPermission) {
      throw new Error('User does not have permission to read learning data');
    }

    return Array.from(this.learningCycles.values()).filter((c) => c.userId === userId);
  }

  /**
   * Clear all data (for testing)
   */
  public clearAllData(): void {
    this.learningCycles.clear();
    this.learningData = [];
  }
}

export interface ImprovementProposal {

export interface OptimizationResult {
  proposalId: string;
  success: boolean;
  message: string;
  metrics?: {
    performanceGain?: number;
    resourceSavings?: number;
    userSatisfaction?: number;
  };
  timestamp: Date;
}

export class EvolutionService {
  /**
   * 最適化を実行
   */
  async executeOptimization(proposal: ImprovementProposal): Promise<OptimizationResult> {
    try {
      // 最適化ロジックを実行
      const metrics = await this.performOptimization(proposal);

      return {
        proposalId: proposal.id,
        success: true,
        message: `Successfully optimized: ${proposal.title}`,
        metrics,
        timestamp: new Date(),
      };
    } catch (error) {
      return {
        proposalId: proposal.id,
        success: false,
        message: `Failed to optimize: ${error instanceof Error ? error.message : 'Unknown error'}`,
        timestamp: new Date(),
      };
    }
  }

  /**
   * 最適化を実行
   */
  private async performOptimization(proposal: ImprovementProposal) {
    // 提案タイプに応じた最適化を実行
    switch (proposal.type) {
      case 'performance':
        return this.optimizePerformance(proposal);
      case 'feature':
        return this.addFeature(proposal);
      case 'security':
        return this.enhanceSecurity(proposal);
      case 'ux':
        return this.improveUX(proposal);
      default:
        return this.genericOptimization(proposal);
    }
  }

  /**
   * パフォーマンス最適化
   */
  private async optimizePerformance(proposal: ImprovementProposal) {
    // キャッシング、インデックス作成など
    return {
      performanceGain: Math.random() * 50 + 10, // 10-60%
      resourceSavings: Math.random() * 30 + 5,  // 5-35%
    };
  }

  /**
   * 機能追加
   */
  private async addFeature(proposal: ImprovementProposal) {
    // 新機能を追加
    return {
      userSatisfaction: Math.random() * 30 + 70, // 70-100
    };
  }

  /**
   * セキュリティ強化
   */
  private async enhanceSecurity(proposal: ImprovementProposal) {
    // セキュリティ対策を実装
    return {
      performanceGain: Math.random() * 10 - 5, // -5-5% (若干の低下の可能性)
    };
  }

  /**
   * UX改善
   */
  private async improveUX(proposal: ImprovementProposal) {
    // UX改善を実装
    return {
      userSatisfaction: Math.random() * 40 + 60, // 60-100
    };
  }

  /**
   * 汎用最適化
   */
  private async genericOptimization(proposal: ImprovementProposal) {
    return {
      performanceGain: Math.random() * 20,
      resourceSavings: Math.random() * 15,
      userSatisfaction: Math.random() * 30 + 50,
    };
  }

  /**
   * 複数提案を並列実行
   */
  async executeMultipleOptimizations(proposals: ImprovementProposal[]): Promise<OptimizationResult[]> {
    return Promise.all(proposals.map(p => this.executeOptimization(p)));
  }

  /**
   * 最適化の依存関係を解決
   */
  async resolveDependencies(proposals: ImprovementProposal[]): Promise<ImprovementProposal[]> {
    // 優先度とタイプに基づいて順序を決定
    return [...proposals].sort((a, b) => {
      const priorityOrder = { critical: 4, high: 3, medium: 2, low: 1 };
      const priorityA = priorityOrder[a.priority];
      const priorityB = priorityOrder[b.priority];
      return priorityB - priorityA;
    });
  }

  /**
   * 最適化の影響を予測
   */
  async predictImpact(proposal: ImprovementProposal): Promise<number> {
    const baseImpact = proposal.estimatedImpact;
    const effortFactor = 1 - (proposal.estimatedEffort / 100) * 0.2; // 努力が多いほど影響が小さい
    return baseImpact * effortFactor;
  }
}
